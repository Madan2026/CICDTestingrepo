pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }
        stage('Docker Build') {
            steps {
                bat 'docker build -t my-app .'
            }
        }
        stage('Deploy') {
            steps {
                bat 'docker run -d -p 8092:80 my-app'
            }
        }
    }
}
